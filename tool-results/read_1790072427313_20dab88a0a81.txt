  1040→function lerpColor(a, b, t) {
  1041→  const ah = parseInt(a.slice(1), 16), bh = parseInt(b.slice(1), 16);
  1042→  const ar = (ah >> 16) & 255, ag = (ah >> 8) & 255, ab = ah & 255;
  1043→  const br = (bh >> 16) & 255, bg = (bh >> 8) & 255, bb = bh & 255;
  1044→  const rr = Math.round(ar + (br - ar) * t), rg = Math.round(ag + (bg - ag) * t), rb = Math.round(ab + (bb - ab) * t);
  1045→  return '#' + ((1 << 24) + (rr << 16) + (rg << 8) + rb).toString(16).slice(1);
  1046→}

  1082→          function wdDistSq(p,a,b){
  1083→            const x=p[0]-a[0],y=p[1]-a[1],dx=b[0]-a[0],dy=b[1]-a[1];
  1084→            if(dx===0&&dy===0)return x*x+y*y;

  1086→            const qx=a[0]+t*dx,qy=a[1]+t*dy; return (x-qx)*(x-qx)+(y-qy)*(y-qy);

  1091→            const stack=[[0,points.length-1]];

  1094→              if(idx>=0){keep[idx]=1;stack.push([a,idx],[idx,b])}
